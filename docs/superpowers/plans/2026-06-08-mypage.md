> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 로그인 유저가 지갑 요약·포인트 내역·내가 쓴 리뷰(보기·삭제)를 보는 마이페이지(`/mypage`)를 만든다.

// 포인트 내역 누적 잔액 계산 (순수 함수).
// 입력: 오래된 순 정렬된 [{ amount, ... }]. 출력: 각 항목에 balance(그 시점 누적 잔액) 추가한 새 배열.
export function computeRunningBalance(rowsAsc) {
  let running = 0
  return rowsAsc.map((row) => {
    running += row.amount
    return { ...row, balance: running }
  })
}

import { supabase } from '../supabase'

// 포인트 내역 — 오래된 순(잔액 누적 계산용). 리뷰 사유는 근장명까지 중첩조회.
export async function fetchPointHistory(userId) {
  const { data, error } = await supabase
    .from('points')
    .select('amount, reason, created_at, review:reviews ( worksite:worksites ( organization_name, workplace_name ) )')
    .eq('user_id', userId)
    .order('created_at', { ascending: true })
  if (error) throw error
  return data ?? []
}

// 내가 쓴 리뷰 — 최신순, 근장명 포함.
export async function fetchMyReviews(userId) {
  const { data, error } = await supabase
    .from('reviews')
    .select('id, worksite_id, work_year, work_semester, opinion, work_content, created_at, worksite:worksites ( organization_name, workplace_name )')
    .eq('user_id', userId)
    .is('deleted_at', null)
    .order('created_at', { ascending: false })
  if (error) throw error
  return data ?? []
}
```

const REASON_LABELS = {
  review_write: '리뷰 작성',
  review_delete: '리뷰 삭제(롤백)',
  ad_view: '광고 시청',
  signup_bonus: '가입 보너스',
  pass_purchase: '열람권 교환',
  pass_refund: '열람권 환불',

function worksiteName(ws) {
  if (!ws) return ''
  return [ws.organization_name, ws.workplace_name].filter(Boolean).join(' - ')
}

function reasonText(row) {
  const label = REASON_LABELS[row.reason] ?? row.reason
  const name = worksiteName(row.review?.worksite)
  return name ? `${label} — ${name}` : label
}

function fmtDateTime(ts) {
  if (!ts) return ''
  const d = new Date(ts)
  const p = (n) => String(n).padStart(2, '0')
  return `${d.getFullYear()}.${p(d.getMonth() + 1)}.${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}`
}

function fmtExpiry(ts) {
  if (!ts) return '없음'
  const d = new Date(ts)
  if (d <= new Date()) return '만료됨'
  const p = (n) => String(n).padStart(2, '0')
  return `${d.getFullYear()}.${p(d.getMonth() + 1)}.${p(d.getDate())}`
}

function stockOf(passBalances, days) {
  return Number(passBalances?.[String(days)] ?? 0)
}

function preview(text, n = 60) {
  const t = (text ?? '').trim()
  return t.length > n ? t.slice(0, n) + '...' : t
}

function errorText(e) {
  if (e.message?.includes('INSUFFICIENT_POINTS')) return '포인트가 부족해 삭제할 수 없어요. (광고 시청 등으로 충전 후 삭제 가능)'
  if (e.message?.includes('NO_PASS_STOCK')) return '보유한 열람권이 없어요.'
  return '처리 중 오류가 발생했어요: ' + e.message
}

            ? <PointHistoryTab rows={pointRows} />
            : <MyReviewsTab reviews={reviews} busy={busy} onDelete={removeReview} onFind={() => navigate('/search')} />}
        </main>
      </div>

      {confirm && (
        <ConfirmModal
          message={confirm.message}
          confirmText={confirm.confirmText}
          onConfirm={() => { const a = confirm.action; setConfirm(null); runAction(a) }}
          onCancel={() => setConfirm(null)}
        />
      )}
    </div>
  )
}
```
